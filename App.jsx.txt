export default function App() {
  const [currentLang, setCurrentLang] = useState('de');
  const t = TRANSLATIONS[currentLang];

  // Grup Siparişi Durumu (Supabase entegrasyonu ile senkronize edilir)
  const [groupSession, setGroupSession] = useState({
    id: "group-vechta-01",
    isSubmitted: false,
    submittedBy: null,
    members: [
      {
        name: "Haci",
        items: [
          { qty: 1, name: "Kalb Kebap", price: 8.00 },
          { qty: 1, name: "Vegetarische Tasche", price: 7.00 }
        ],
        total: 15.00,
        wheelReward: null
      },
      {
        name: "Lütfü",
        items: [
          { qty: 1, name: "Kebap Spezial Weichkäse", price: 8.50 },
          { qty: 1, name: "Kebap Hollandaise Soße", price: 8.50 },
          { qty: 1, name: "Pizza Kebap (groß)", extras: ["Pilzen"], price: 10.00 },
          { qty: 1, name: "Pizza Rastaa (groß)", extras: ["Ananas"], price: 12.00 }
        ],
        total: 39.00,
        wheelReward: { title: "Gratis Getränk", code: "ANXKXP" }
      },
      {
        name: "Şikici",
        items: [
          { qty: 1, name: "Kalb Kebap", price: 8.00 },
          { qty: 1, name: "Kebap Spezial Weichkäse", price: 8.50 },
          { qty: 1, name: "Kebap Hollandaise Soße", price: 8.50 },
          { qty: 1, name: "Jumbo Kebap", price: 9.00 },
          { qty: 1, name: "Bodrum Gemüse Kebap", price: 9.00 },
          { qty: 1, name: "Vegetarische Tasche", price: 7.00 },
          { qty: 1, name: "Kebap Box", price: 8.00 }
        ],
        total: 58.00,
        wheelReward: { title: "10% Rabatt", code: "TR8821" }
      }
    ]
  });

  // ----------------------------------------------------
  // 1. PIZZA EKSTRALARINI '+' İLE FORMATLAMA METODU
  // ----------------------------------------------------
  const formatItemExtras = (extras) => {
    if (!extras || extras.length === 0) return "";
    // Eskiden '-' kullanılan yapı '+' işaretiyle değiştirildi
    return ` + ${extras.join(', ')}`;
  };

  // ----------------------------------------------------
  // 2. WHATSAPP MESAJ METNİ VE ÇARKI FELEK KAZANANLARI
  // ----------------------------------------------------
  const generateWhatsAppMessage = (groupMembers = []) => {
    let msg = "";

    if (groupMembers.length > 0) {
      // Grup Siparişi İçeriği
      groupMembers.forEach(member => {
        msg += `👤 ${member.name}:\n`;
        member.items.forEach(item => {
          const extrasText = item.extras ? formatItemExtras(item.extras) : "";
          msg += ` • ${item.qty}x ${item.name}${extrasText} (${item.price.toFixed(2).replace('.', ',')} €)\n`;
        });
        msg += `\n`;
      });
    }

    // Toplam Hesaplama
    const grandTotal = groupMembers.reduce((sum, m) => sum + m.total, 0);
    msg += `${t.total}: ${grandTotal.toFixed(2).replace('.', ',')} €\n\n`;

    // Çarkıfelek Ödüllerini Mesajın En Altına Ekleme
    const wheelWinners = groupMembers.filter(m => m.wheelReward);
    if (wheelWinners.length > 0) {
      msg += `${t.wheelGifts}\n`;
      wheelWinners.forEach(w => {
        msg += ` • ${w.name}: ${w.wheelReward.title} (Code: ${w.wheelReward.code})\n`;
      });
      msg += `\n`;
    }

    msg += `(Abholung, keine Lieferung) Bitte sagt uns, wann es abholbereit ist. Danke!`;
    return encodeURIComponent(msg);
  };

  // ----------------------------------------------------
  // 3. SİPARİŞİ GÖNDERME VE DİĞERLERİNE KİLİTLEME
  // ----------------------------------------------------
  const handleSendGroupOrder = (senderName) => {
    // Siparişi gönderen kişiyi kaydet ve grubu kilitle
    setGroupSession(prev => ({
      ...prev,
      isSubmitted: true,
      submittedBy: senderName
    }));

    const message = generateWhatsAppMessage(groupSession.members);
    window.open(`https://wa.me/4915210804759?text=${message}`, '_blank');
  };

  return (
    <div className="min-h-screen bg-neutral-900 text-neutral-100 font-sans">
      
      {/* HEADER / ÜST BÖLÜM & DİL SEÇİMİ */}
      <header className="sticky top-0 z-50 bg-neutral-900/95 backdrop-blur border-b border-neutral-800 p-4">
        <div className="max-w-4xl mx-auto flex flex-wrap justify-between items-center gap-3">
          <div className="flex items-center gap-2">
            <span className="text-2xl">🥙</span>
            <div>
              <h1 className="text-xl font-bold text-orange-500 leading-none">Bodrum Kebap</h1>
              <span className="text-xs text-neutral-400">Vechta • Oyther Str. 37</span>
            </div>
          </div>

          {/* DİL SEÇENEKLERİ (DE, TR, EN, RO, NL) */}
          <div className="flex items-center gap-1.5 bg-neutral-800 p-1 rounded-xl border border-neutral-700">
            {Object.keys(TRANSLATIONS).map((langKey) => (
              <button
                key={langKey}
                onClick={() => setCurrentLang(langKey)}
                className={`px-2.5 py-1 text-xs font-semibold rounded-lg transition-all flex items-center gap-1 ${
                  currentLang === langKey 
                    ? 'bg-orange-500 text-white shadow-md' 
                    : 'text-neutral-400 hover:text-white hover:bg-neutral-700/50'
                }`}
              >
                <span>{TRANSLATIONS[langKey].flag}</span>
                <span>{langKey.toUpperCase()}</span>
              </button>
            ))}
          </div>
        </div>
      </header>

      {/* İÇERİK BÖLÜMÜ */}
      <main className="max-w-2xl mx-auto p-4 space-y-6">

        {/* GRUP SİPARİŞİ KARTI */}
        <div className="bg-neutral-800/80 border border-neutral-700 rounded-2xl p-5 shadow-xl">
          <div className="flex justify-between items-center mb-4 pb-3 border-b border-neutral-700">
            <h2 className="text-lg font-bold text-orange-400 flex items-center gap-2">
              <span>👥</span> {t.groupOrder}
            </h2>
            <span className="text-xs bg-orange-500/10 text-orange-400 px-2.5 py-1 rounded-full border border-orange-500/20 font-medium">
              Aktif Oturum
            </span>
          </div>

          {/* SİPARİŞ KİLİT BİLDİRİMİ VEYA GÖNDERME BUTONU */}
          {groupSession.isSubmitted ? (
            <div className="bg-emerald-950/90 border border-emerald-500/60 p-4 rounded-xl text-center space-y-1 my-4">
              <div className="text-3xl">✅</div>
              <p className="text-emerald-200 font-semibold text-sm">
                {t.orderSubmitted.replace('{name}', groupSession.submittedBy)}
              </p>
            </div>
          ) : (
            <div className="space-y-4">
              {/* Grup Sipariş Listesi Önizlemesi */}
              <div className="space-y-4 text-sm">
                {groupSession.members.map((member, idx) => (
                  <div key={idx} className="bg-neutral-900/60 p-3 rounded-xl border border-neutral-800">
                    <div className="font-bold text-neutral-300 mb-1 flex items-center gap-1.5">
                      <span>👤</span> {member.name}:
                    </div>
                    <ul className="space-y-1 text-neutral-400 pl-5 list-disc">
                      {member.items.map((item, i) => (
                        <li key={i}>
                          {item.qty}x {item.name}
                          {item.extras && <span className="text-orange-400 font-medium">{formatItemExtras(item.extras)}</span>}
                          {" "}({item.price.toFixed(2).replace('.', ',')} €)
                        </li>
                      ))}
                    </ul>
                  </div>
                ))}
              </div>

              {/* Sipariş Gönder Butonu */}
              <button
                onClick={() => handleSendGroupOrder("Haci")}
                className="w-full py-3.5 bg-emerald-600 hover:bg-emerald-500 text-white font-bold rounded-xl transition shadow-lg shadow-emerald-950/50 flex items-center justify-center gap-2 text-base"
              >
                <span>💬</span>
                {t.sendWhatsapp}
              </button>
            </div>
          )}
        </div>

      </main>
    </div>
  );
}
