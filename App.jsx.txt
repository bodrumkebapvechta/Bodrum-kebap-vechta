const LANGUAGES = [
  { code: 'de', flag: '🇩🇪', label: 'DE' },
  { code: 'tr', flag: '🇹🇷', label: 'TR' },
  { code: 'en', flag: '🇬🇧', label: 'EN' },
  { code: 'ro', flag: '🇷🇴', label: 'RO' },
  { code: 'nl', flag: '🇳🇱', label: 'NL' }
];

export default function App() {
  const [currentLang, setCurrentLang] = useState('de');
  const t = TRANSLATIONS[currentLang] || TRANSLATIONS.de;

  const [groupSession, setGroupSession] = useState({
    id: "group-vechta-01",
    isSubmitted: false,
    submittedBy: null,
    members: [
      {
        name: "Haci",
        items: [
          { qty: 1, name: "Kalb Kebap", price: 8.00 },
          { qty: 1, name: "Vegetarische Tasche", price: 7.00 }
        ],
        total: 15.00,
        wheelReward: null
      },
      {
        name: "Lütfü",
        items: [
          { qty: 1, name: "Kebap Spezial Weichkäse", price: 8.50 },
          { qty: 1, name: "Pizza Kebap (groß)", extras: ["Pilzen"], price: 10.00 }
        ],
        total: 18.50,
        wheelReward: { title: "Gratis Getränk", code: "ANXKXP" }
      }
    ]
  });

  const formatItemExtras = (extras) => {
    if (!extras || extras.length === 0) return "";
    return ` + ${extras.join(', ')}`;
  };

  const handleSendGroupOrder = (senderName) => {
    setGroupSession(prev => ({
      ...prev,
      isSubmitted: true,
      submittedBy: senderName
    }));

    let msg = `*${t.groupOrder}*\n\n`;
    groupSession.members.forEach(member => {
      msg += `👤 *${member.name}:*\n`;
      member.items.forEach(item => {
        const extrasText = item.extras ? formatItemExtras(item.extras) : "";
        msg += ` • ${item.qty}x ${item.name}${extrasText} (${item.price.toFixed(2).replace('.', ',')} €)\n`;
      });
      msg += `\n`;
    });

    const grandTotal = groupSession.members.reduce((sum, m) => sum + m.total, 0);
    msg += `*${t.total}: ${grandTotal.toFixed(2).replace('.', ',')} €*\n\n`;

    const wheelWinners = groupSession.members.filter(m => m.wheelReward);
    if (wheelWinners.length > 0) {
      msg += `*${t.wheelGifts}*\n`;
      wheelWinners.forEach(w => {
        msg += ` • ${w.name}: ${w.wheelReward.title} (Code: ${w.wheelReward.code})\n`;
      });
    }

    window.open(`https://wa.me/4915210804759?text=${encodeURIComponent(msg)}`, '_blank');
  };

  return (
    <div className="min-h-screen bg-neutral-900 text-neutral-100 font-sans pb-10">
      {/* HEADER & DİL SEÇİMİ */}
      <header className="sticky top-0 z-50 bg-neutral-900/95 backdrop-blur border-b border-neutral-800 p-4 shadow-lg">
        <div className="max-w-4xl mx-auto flex flex-wrap justify-between items-center gap-3">
          <div className="flex items-center gap-2">
            <span className="text-3xl">🥙</span>
            <div>
              <h1 className="text-xl font-bold text-orange-500 leading-none">{t.title}</h1>
              <span className="text-xs text-neutral-400">{t.subtitle}</span>
            </div>
          </div>

          {/* DİL DESTEĞİ BUTONLARI */}
          <div className="flex items-center gap-1 bg-neutral-800 p-1.5 rounded-xl border border-neutral-700">
            {LANGUAGES.map((lang) => (
              <button
                key={lang.code}
                onClick={() => setCurrentLang(lang.code)}
                className={`px-2.5 py-1 text-xs font-bold rounded-lg transition-all flex items-center gap-1 ${
                  currentLang === lang.code
                    ? 'bg-orange-500 text-white shadow-md scale-105'
                    : 'text-neutral-400 hover:text-white hover:bg-neutral-700/50'
                }`}
              >
                <span>{lang.flag}</span>
                <span>{lang.label}</span>
              </button>
            ))}
          </div>
        </div>
      </header>

      {/* İÇERİK BÖLÜMÜ */}
      <main className="max-w-2xl mx-auto p-4 space-y-6 mt-2">
        {/* GRUP SİPARİŞİ PANENİ */}
        <div className="bg-neutral-800/90 border border-neutral-700 rounded-2xl p-5 shadow-2xl">
          <div className="flex justify-between items-center mb-4 pb-3 border-b border-neutral-700">
            <h2 className="text-lg font-bold text-orange-400 flex items-center gap-2">
              <span>👥</span> {t.groupOrder}
            </h2>
            <span className="text-xs bg-orange-500/10 text-orange-400 px-3 py-1 rounded-full border border-orange-500/20 font-semibold">
              {t.activeSession}
            </span>
          </div>

          {groupSession.isSubmitted ? (
            <div className="bg-emerald-950/90 border border-emerald-500/60 p-4 rounded-xl text-center space-y-1 my-4">
              <div className="text-3xl">✅</div>
              <p className="text-emerald-200 font-semibold text-sm">
                {t.orderSubmitted.replace('{name}', groupSession.submittedBy)}
              </p>
            </div>
          ) : (
            <div className="space-y-4">
              <div className="space-y-3 text-sm">
                {groupSession.members.map((member, idx) => (
                  <div key={idx} className="bg-neutral-900/80 p-3.5 rounded-xl border border-neutral-800">
                    <div className="font-bold text-neutral-200 mb-1.5 flex items-center gap-1.5">
                      <span>👤</span> {member.name}:
                    </div>
                    <ul className="space-y-1 text-neutral-400 pl-5 list-disc">
                      {member.items.map((item, i) => (
                        <li key={i}>
                          {item.qty}x {item.name}
                          {item.extras && <span className="text-orange-400 font-medium">{formatItemExtras(item.extras)}</span>}
                          {" "}({item.price.toFixed(2).replace('.', ',')} €)
                        </li>
                      ))}
                    </ul>
                  </div>
                ))}
              </div>

              <button
                onClick={() => handleSendGroupOrder("Haci")}
                className="w-full py-3.5 bg-emerald-600 hover:bg-emerald-500 text-white font-bold rounded-xl transition-all shadow-lg shadow-emerald-950/50 flex items-center justify-center gap-2 text-base active:scale-95"
              >
                <span>💬</span>
                {t.sendWhatsapp}
              </button>
            </div>
          )}
        </div>
      </main>
    </div>
  );
}
